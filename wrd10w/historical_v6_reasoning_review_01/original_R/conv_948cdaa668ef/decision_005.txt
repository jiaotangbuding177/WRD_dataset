from openpyxl import Workbook, load_workbook

def classify(summary):
    s = summary or ''
    if any(k in s for k in ('利息','收息','结息','存款利息')):
        return 'interest'
    if any(k in s for k in ('手续费','服务费','工本费','网上企业银行','网银')):
        return 'fee'
    if any(k in s for k in ('出资','投资款','实收资本','股东')):
        return 'capital'
    return 'expense'

def _norm_key(s):
    return str(s) if s is not None else ''

def read_journal(path):
    wb = load_workbook(path, data_only=True)
    ws = wb.active
    rows = [[c for c in r] for r in ws.iter_rows(values_only=True)]
    if not rows: return []
    header = [ _norm_key(h) for h in rows[0] ]
    def find(cols):
        for i,h in enumerate(header):
            if any(c in h for c in cols): return i
        return None
    i_date = find(['日期','时间'])
    i_sum  = find(['摘要','备注','用途','说明'])
    i_in   = find(['借方','收入','进账','存入'])
    i_out  = find(['贷方','支出','出账','付出'])
    if i_sum is None:
        raise ValueError('summary column not found: %r' % header)
    out=[]
    for r in rows[1:]:
        if r is None or all(v is None or str(v).strip()=='' for v in r): continue
        def g(i):
            return r[i] if (i is not None and i < len(r)) else None
        out.append({
            'date': str(g(i_date)) if g(i_date) is not None else '',
            'summary': str(g(i_sum)) if g(i_sum) is not None else '',
            'debit': float(g(i_in) or 0),
            'credit': float(g(i_out) or 0),
        })
    return out
