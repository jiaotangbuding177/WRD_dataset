```python
import os, json

ws = wb.active
ws.title = "银行日记账"
ws.append(["日期", "摘要", "借方", "贷方", "余额"])
rows = [
    ["2025-08-07", "收到股东出资款", 30000, None, 30000],
    ["2025-10-09", "支付一力行服务费", None, 5500, 24500],
    ["2025-11-02", "网银跨行本地手续费", None, 1.76, 24498.24],
    ["2025-12-21", "收息", 0.14, None, 24498.38],
    ["2026-03-21", "收息", 0.13, None, 24498.51],
    ["2026-06-21", "收息", 0.29, None, 24498.80],
]
for r in rows:
    ws.append(r)
wb.save(inp)

# Reusable function
def classify(summary):
    s = str(summary)
    if "收息" in s or "利息" in s:
        return "interest"
    if "手续费" in s or "服务费" in s:
        return "fee"
    if "出资" in s or "投资款" in s:
        return "capital"
    return "other"

def build_vouchers(rows):
    vouchers = []
    for date, summary, debit, credit, balance in rows:
        cat = classify(summary)
        if cat == "interest":
            amt = float(debit)
            vouchers.append({"date": date, "summary": summary,
                             "entries": [("银行存款", amt), ("财务费用—手续费", -amt)]})
        elif cat == "fee":
            amt = float(credit)
            vouchers.append({"date": date, "summary": summary,
                             "entries": [("财务费用—手续费", amt), ("银行存款", -amt)]})
        elif cat == "capital":
            amt = float(debit)
            vouchers.append({"date": date, "summary": summary,
                             "entries": [("银行存款", amt), ("实收资本", -amt)]})
        else:
            amt = float(debit if debit else credit)
            vouchers.append({"date": date, "summary": summary,
                             "entries": [("银行存款", amt), ("其他", -amt)]})
    return vouchers

# read back
wb2 = load_workbook(inp)
ws2 = wb2.active
data = list(ws2.iter_rows(min_row=2, values_only=True))
vouchers = build_vouchers(data)

# checks
assert len(vouchers) == len(rows), "voucher count mismatch"
for v in vouchers:
    total = sum(a for _, a in v["entries"])
    assert abs(total) < 1e-9, f"unbalanced voucher {v}"
interest = [v for v in vouchers if "收息" in v["summary"]]
assert len(interest) == 3, "interest count"
for v in interest:
    assert v["entries"][1][1] < 0, "interest must offset financial expense with negative"
    assert v["entries"][0][0] == "银行存款"
    assert v["entries"][1][0] == "财务费用—手续费"

# output
outp = os.path.join(out_dir, "vouchers.json")
with open(outp, "w", encoding="utf-8") as f:
    json.dump(vouchers, f, ensure_ascii=False, indent=2)
print("OK", len(vouchers))
```

```python
import os, json
from openpyxl import Workbook, load_workbook

WS = os.environ.get("WORKSPACE", ".")
OUT = os.path.join(WS, "outputs")
os.makedirs(OUT, exist_ok=True)

def classify(summary):
    s = str(summary)
    if "收息" in s or "利息" in s:
        return "interest"
    if "手续费" in s or "服务费" in s:
        return "fee"
    if "出资" in s or "投资款" in s:
        return "capital"
    return "other"

def build_vouchers(rows):
    vouchers = []
    for row in rows:
        date, summary, debit, credit = row[0], row[1], row[2], row[3]
        cat = classify(summary)
        if cat == "interest":
            amt = float(debit)
            entries = [("银行存款", amt), ("财务费用—手续费", -amt)]
        elif cat == "fee":
            amt = float(credit)
            entries = [("财务费用—手续费", amt), ("银行存款", -amt)]
        elif cat == "capital":
            amt = float(debit)
            entries = [("银行存款", amt), ("实收资本", -amt)]
        else:
            amt = float(debit if debit is not None else credit)
            entries = [("银行存款", amt), ("其他", -amt)]
        vouchers.append({"date": str(date), "summary": str(summary), "entries": entries})
    return vouchers

def process(input_path, output_path):
    wb = load_workbook(input_path)
    ws = wb.active
    rows = [r for r in ws.iter_rows(min_row=2, values_only=True) if r[0] is not None]
    vouchers = build_vouchers(rows)
    with open(output_path, "w", encoding="utf-8") as f:
        json.dump(vouchers, f, ensure_ascii=False, indent=2)
    return vouchers, rows

# --- self test with synthetic data ---
inp = os.path.join(OUT, "synthetic_bank_journal.xlsx")

    ["2025-08-07", "收到股东出资款", 30000, None, 30000],
    ["2025-10-09", "支付一力行服务费", None, 5500, 24500],
    ["2025-11-02", "网银跨行本地手续费", None, 1.76, 24498.24],
    ["2025-12-21", "收息", 0.14, None, 24498.38],
    ["2026-03-21", "收息", 0.13, None, 24498.51],
    ["2026-06-21", "收息", 0.29, None, 24498.80],
]
for r in data:
    ws.append(r)
wb.save(inp)

outp = os.path.join(OUT, "vouchers.json")
vouchers, rows = process(inp, outp)

assert len(vouchers) == len(data), "voucher count must equal transaction count"
for v in vouchers:
    assert abs(sum(a for _, a in v["entries"])) < 1e-9, f"unbalanced: {v}"
```
