import os, zipfile
from xml.sax.saxutils import escape

WS = os.getcwd()
OUT = os.path.join("outputs", "test_vouchers.docx")
os.makedirs("outputs", exist_ok=True)

vouchers = [
    {"date":"2025-09-21","rows":[("收银行存款利息","银行存款", 4.23, None),
                                 ("", "财务费用—手续费", -4.23, None)]},
    {"date":"2025-08-01","rows":[("支付银行手续费","财务费用—手续费", 25.0, None),
                                 ("", "银行存款", None, 25.0)]},
]

def cell(v):
    if v is None: return "<w:tc><w:p/></w:tc>"
    if isinstance(v,(int,float)): txt = ("%g"%v) if v==int(v) else ("%.2f"%v)
    else: txt = v
    return "<w:tc><w:p><w:r><w:t xml:space=\"preserve\">%s</w:t></w:r></w:p></w:tc>"%escape(txt)

rows_xml = []
for v in vouchers:
    for (summ, subj, dr, cr) in v["rows"]:
        rows_xml.append("<w:tr>%s%s%s%s</w:tr>"%(cell(summ),cell(subj),cell(dr),cell(cr)))
doc_xml = '<?xml version="1.0"?><w:document xmlns:w="http://schemas.openxmlformats.org/wordprocessingml/2006/main"><w:body><w:tbl>%s</w:tbl></w:body></w:document>'%"".join(rows_xml)

content_types = '<?xml version="1.0"?><Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types"><Default Extension="xml" ContentType="application/xml"/><Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/><Override PartName="/word/document.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.document.main+xml"/></Types>'
rels = '<?xml version="1.0"?><Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships"><Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument" Target="word/document.xml"/></Relationships>'

with zipfile.ZipFile(OUT,"w",zipfile.ZIP_DEFLATED) as z:
    z.writestr("[Content_Types].xml", content_types)
    z.writestr("_rels/.rels", rels)
    z.writestr("word/document.xml", doc_xml)

# readback verification
assert os.path.exists(OUT) and os.path.getsize(OUT)>0
with zipfile.ZipFile(OUT) as z:
    names = z.namelist()
    assert "word/document.xml" in names
    xml = z.read("word/document.xml").decode("utf-8")

assert "财务费用—手续费" in xml
assert "财务费用—利息收入" not in xml

assert "4.23" in xml
# balance: interest voucher dr total 4.23 + (-4.23) = 0 == cr 0... hmm net zero
