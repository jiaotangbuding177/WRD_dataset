def post_money(increase, ratio):
    assert 0 < ratio < 1, "ratio must be in (0,1)"

    return post, pre

    assert pre_registered_capital <= pre + 1e-6, "pre registered capital exceeds pre-money valuation"
    delta = increase * pre_registered_capital / pre
    reserve = increase - delta
