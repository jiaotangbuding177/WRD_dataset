    except Exception as e:
        mods[m]=False

NS="{http://schemas.openxmlformats.org/spreadsheetml/2006/main}"
z=zipfile.ZipFile(path)
names=z.namelist()
shared=[]
if "xl/sharedStrings.xml" in names:
    r=ET.fromstring(z.read("xl/sharedStrings.xml"))
    for si in r.findall(NS+"si"):
        shared.append("".join(t.text or "" for t in si.iter(NS+"t")))
wb=ET.fromstring(z.read("xl/workbook.xml"))
rels=ET.fromstring(z.read("xl/_rels/workbook.xml.rels"))
relmap={rel.get("Id"):rel.get("Target") for rel in rels}
sheets=[]
for s in wb.iter(NS+"sheet"):
    rid=s.get("{http://schemas.openxmlformats.org/officeDocument/2006/relationships}id")
    sheets.append((s.get("name"), relmap.get(rid)))
print("SHEETS", sheets)
assert sheets, "no sheets"

def col_letter(ref):
    return "".join(c for c in ref if c.isalpha())

for name,target in sheets:
    t = target if target.startswith("xl/") else "xl/"+target.lstrip("/")
    if t not in names:
        # try
        t = "xl/worksheets/"+os.path.basename(target)
    root=ET.fromstring(z.read(t))
    rows=[]
    for row in root.iter(NS+"row"):
        cells={}
        for c in row.findall(NS+"c"):
            ref=c.get("r"); typ=c.get("t")
            v=c.find(NS+"v")
            if typ=="s" and v is not None:
                val=shared[int(v.text)]
            elif typ=="inlineStr":
                val="".join(x.text or "" for x in c.iter(NS+"t"))
            else:
                val=v.text if v is not None else None
            cells[col_letter(ref)]=val
        rows.append(cells)
        if len(rows)>=6: break
    print("SHEET", name, "first_rows", json.dumps(rows, ensure_ascii=False))
