import os, json

def post_money(increase, ratio):
    assert 0 < ratio < 1, "ratio must be in (0,1)"
    assert increase > 0
    post = increase / ratio
    pre = post - increase
    return post, pre

def split_capital(increase, ratio, pre_registered_capital):
    post, pre = post_money(increase, ratio)
    assert pre_registered_capital <= pre + 1e-6, "pre registered capital exceeds pre-money valuation"
    delta = increase * pre_registered_capital / pre
    reserve = increase - delta
    return delta, reserve

def run(increase, ratio, pre_registered_capital=None):
    post, pre = post_money(increase, ratio)
    assert abs(increase / post - ratio) < 1e-9
    result = {"post_money": round(post, 2), "pre_money": round(pre, 2)}
    if pre_registered_capital is None:
        result["split"] = None
        result["note"] = "need pre_registered_capital; constraint: pre_registered_capital <= pre_money"
    else:
        delta, reserve = split_capital(increase, ratio, pre_registered_capital)
        assert abs(delta + reserve - increase) < 1e-6
        assert abs(delta / (pre_registered_capital + delta) - ratio) < 1e-9
        result["split"] = {"registered_capital": round(delta, 2), "capital_reserve": round(reserve, 2)}
    return result

# instance test
r = run(480.0, 0.142012)
